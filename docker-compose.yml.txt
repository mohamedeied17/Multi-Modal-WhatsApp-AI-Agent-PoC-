version: '3.8'

services:
  postgres:
    image: postgres:15
    container_name: postgres_db
    environment:
      - POSTGRES_USER=myuser
      - POSTGRES_PASSWORD=mypassword
      - POSTGRES_DB=evolution
    ports:
      - "5432:5432"
    restart: always

  redis:
    image: redis:latest
    container_name: redis_cache
    ports:
      - "6379:6379"
    restart: always

  evolution-api:
    image: evoapicloud/evolution-api:latest
    container_name: evolution_api
    environment:
      - SERVER_URL=http://localhost:8080
      - DOCKER_ENV=true
      - DATABASE_PROVIDER=postgresql
      - DATABASE_CONNECTION_URI=postgresql://myuser:mypassword@postgres:5432/evolution
      - CACHE_REDIS_URI=redis://redis:6379
    ports:
      - "8080:8080"
    depends_on:
      - postgres
      - redis
    restart: always

  n8n:
    image: n8nio/n8n
    container_name: n8n_engine
    ports:
      - "5678:5678"
    environment:
      - N8N_HOST=localhost
      - N8N_PORT=5678
      - N8N_PROTOCOL=http
      - NODE_ENV=production
    restart: always
